from typing import assert_type

import numpy as np

from bokeh.core.properties import field
from bokeh.models import ColumnDataSource, Line, MultiLine, PathPlacement, Scatter, Segment
from bokeh.models.annotations.arrows import ArrowHead
from bokeh.models.glyph import Glyph
from bokeh.models.graphics import Marking
from bokeh.models.plots import Plot
from bokeh.models.renderers import GlyphRenderer
from bokeh.plotting import figure


def plot_helpers(head: ArrowHead) -> None:
    p = figure()
    segment = p.segment()
    assert_type(segment.add_arrowheads(), PathPlacement)
    assert_type(segment.add_arrowheads(at="start", head=head), PathPlacement)
    assert_type(segment.add_arrowheads(at=0.25), PathPlacement)
    assert_type(segment.add_arrowheads(at=[0.25, 0.75]), PathPlacement)
    assert_type(segment.add_arrowheads(at=np.array([0.25, 0.75])), PathPlacement)
    assert_type(segment.add_arrowheads(at=field("placements")), PathPlacement)
    assert_type(segment.add_path_marking(head, at=field("placements")), PathPlacement)
    assert_type(segment.add_path_marking(head, at="end", inset=8), PathPlacement)
    assert_type(segment.add_path_marking(head, at=[0.5]), PathPlacement)
    segment.add_path_marking(head, at="middle", inset=8)  # mypy: call-overload | pyright: reportArgumentType

    p.scatter().add_arrowheads(at="start")  # mypy: misc | pyright: reportAttributeAccessIssue
    p.step().add_arrowheads(at=[0.25, 0.75])  # mypy: misc | pyright: reportAttributeAccessIssue
    p.scatter().add_path_marking(head, at="end")  # mypy: misc | pyright: reportAttributeAccessIssue
    p.step().add_path_marking(head, at=[0.25, 0.75])  # mypy: misc | pyright: reportAttributeAccessIssue
    p.rect().add_path_marking(head, at="end")  # mypy: misc | pyright: reportAttributeAccessIssue
    p.rect().glyph.path_markings  # mypy: attr-defined | pyright: reportAttributeAccessIssue
    assert_type(segment.glyph.path_markings, list[PathPlacement])


def add_glyph_preserves_path_hosts() -> None:
    plot = Plot()
    segment = plot.add_glyph(Segment())
    line = plot.add_glyph(ColumnDataSource(), glyph=Line())
    assert_type(segment, GlyphRenderer[Segment])
    assert_type(line, GlyphRenderer[Line])
    assert_type(segment.add_arrowheads(), PathPlacement)
    assert_type(line.add_arrowheads(at=[0.25, 0.75]), PathPlacement)

    plot.add_glyph(Scatter()).add_arrowheads()  # mypy: misc | pyright: reportAttributeAccessIssue
    plot.add_glyph(ColumnDataSource(), Scatter()).add_arrowheads()  # mypy: misc | pyright: reportAttributeAccessIssue


class CustomLine(Line):
    pass


def reusable(path: GlyphRenderer[Line | MultiLine], custom: GlyphRenderer[CustomLine], marking: Marking) -> None:
    assert_type(path.add_arrowheads(at="middle"), PathPlacement)
    assert_type(custom.add_arrowheads(at=[0.25, 0.75]), PathPlacement)
    assert_type(path.add_path_marking(marking, at="end"), PathPlacement)
    assert_type(custom.add_path_marking(marking, at=[0.25, 0.75]), PathPlacement)


def mixed(renderer: GlyphRenderer[Line | Scatter], marking: Marking) -> None:
    renderer.add_arrowheads()  # mypy: misc | pyright: reportAttributeAccessIssue
    renderer.add_path_marking(marking, at=0.5)  # mypy: misc | pyright: reportAttributeAccessIssue


def general_path_methods(renderer: GlyphRenderer[Glyph], marking: Marking) -> None:
    renderer.add_path_marking(marking, at="end")  # mypy: misc | pyright: reportAttributeAccessIssue
