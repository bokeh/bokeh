from typing import Sequence, assert_type

from bokeh.core.properties import value
from bokeh.model import CallbackPolicy, Committed, Debounce, Immediate, Throttle
from bokeh.models import ContourLabels, CustomJS, Range1d
from bokeh.models.layouts import Column, Row
from bokeh.models.ui.ui_element import UIElement
from bokeh.models.widgets.sliders import Slider
from bokeh.plotting import figure

# Original issue #14412 reproducer and its constructor overloads.
p = figure(width=800, height=200, x_axis_label="x-axis", y_axis_label="y-axis", tools="")
assert_type(p, figure)
assert_type(Range1d(0, 10, bounds="auto"), Range1d)
Range1d(start=0, end=10, bounds=(0, 20))
Range1d(bounds="auto")
slider = Slider(start=0, end=10, value=3)
def update(attr: str, old: object, new: object) -> None: ...
throttle: CallbackPolicy = Throttle(100)
slider.on_change("value", update, policy=throttle)
slider.on_change("value", update, policy=Immediate())
slider.on_change("value", update, policy=Debounce(250))
slider.on_change("value", update, policy=Committed())
slider.js_on_change("value", CustomJS(code=""), policy=Immediate())
slider.js_on_change("value", CustomJS(code=""), policy=Throttle(100))
slider.js_on_change("value", CustomJS(code=""), policy=Debounce(250))
slider.js_on_change("value", CustomJS(code=""), policy=Committed())

# Glyph calls with vectorized and scalar properties.
p.scatter([1, 2], [3, 4], size=10, fill_color="red")
p.scatter(x="x", y="y", marker="circle", size="size", source={"x": [1], "y": [2], "size": [3]})

# Input types can be covariant without weakening the mutable getter type.
rows: list[Row] = [Row()]
columns: list[Column] = [Column()]
Row(Row(), Column(), width=100)
Column(*rows, spacing=5)
children: Sequence[UIElement] = tuple(rows)
row = Row(children=columns)
column = Column(children=rows)
Row(children=children)
Column(children=tuple(rows))
row.children = columns
column.children = rows
column.children = children
assert_type(row.children, list[UIElement])
assert_type(column.children, list[UIElement])

# Hatch pattern abbreviations are accepted like full pattern names (#14937).
p.vbar(x=[1, 2], top=[3, 4], hatch_pattern=["x", "."])
p.vbar(x=[1, 2], top=[3, 4], hatch_pattern=value("x"))
p.vbar(x=[1, 2], top=[3, 4], hatch_pattern=["cross", "dot"])

# Contour labels are editable models with the same styles at construction and through the renderer.
contours = p.contour([0, 1], [0, 1], [[0, 1], [1, 2]], [0.5, 1.5], line_color="black")
assert_type(contours.add_labels(background_fill_color="yellow", background_fill_alpha=0.5, name="levels"), ContourLabels)
assert_type(ContourLabels(background_fill_color=None, background_fill_alpha=0.5, name="levels"), ContourLabels)
