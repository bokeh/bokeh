from bokeh.models import ColumnDataSource, Range1d
from bokeh.models.layouts import Column, Row
from bokeh.models.widgets.sliders import Slider
from bokeh.plotting import figure

figure(unknown_property=1)  # mypy: call-arg; pyright: reportCallIssue
figure(unknown_first=1, unknown_second=2)  # mypy: call-arg,call-arg; pyright: reportCallIssue,reportCallIssue
figure(x_axis_label=123)  # mypy: arg-type; pyright: reportArgumentType
Range1d(0)  # mypy: call-overload; pyright: reportCallIssue
Range1d(bounds="invalid")  # mypy: call-overload; pyright: reportArgumentType
Slider(value="invalid")  # mypy: arg-type; pyright: reportArgumentType
Slider(value_throttled=1)  # mypy: call-arg; pyright: reportCallIssue
Column(children=[ColumnDataSource()])  # mypy: list-item; pyright: reportArgumentType
Column(Row(), children=[])  # mypy: call-overload; pyright: reportCallIssue
column = Column()
column.children = (ColumnDataSource(),)  # mypy: assignment; pyright: reportAttributeAccessIssue
column.children.append(ColumnDataSource())  # mypy: arg-type; pyright: reportArgumentType
p = figure()
p.scatter([1], [2], unknown_property=1)  # mypy: call-arg; pyright: reportCallIssue
p.scatter([1], [2], size="size", fill_alpha="alpha", source={"x": [1]})
