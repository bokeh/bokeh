from bokeh.core.query import IN, OR
from bokeh.document import Document
from bokeh.models.layouts import Column
from bokeh.models.tools import HoverTool, PanTool
from bokeh.plotting import figure

col = Column(figure(name="plot"))
doc = Document()
doc.add_root(col)

# Selectors can be dict displays or dict() calls.
col.select({"type": HoverTool})
col.select(dict(type=HoverTool))
col.select(dict(type={IN: [HoverTool, PanTool]}))
col.select({OR: [dict(type=HoverTool), dict(name="plot")]})
col.select({"tags": ["foo"], OR: [{"name": "plot"}]})
col.select_one(dict(name="plot"))
col.set_select(dict(type=HoverTool), dict(description="hover"))
col.set_select(HoverTool, dict(description="hover"))
doc.select(dict(name="plot"))
doc.select_one(dict(type=PanTool))
doc.set_select(dict(type=HoverTool), dict(description="hover"))

col.select("plot")  # mypy: arg-type; pyright: reportArgumentType
doc.select_one([HoverTool])  # mypy: arg-type; pyright: reportArgumentType
