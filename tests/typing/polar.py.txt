from typing import assert_type

from bokeh.models import PolarMapping, Range1d, Scatter
from bokeh.models.renderers import GlyphRenderer
from bokeh.plotting import figure

p = figure(coordinate_system=PolarMapping(radial_range=Range1d(0, 10)))
assert_type(p.scatter(radius=[2, 4], angle=[0, 90], marker_angle=0.5), GlyphRenderer[Scatter])
