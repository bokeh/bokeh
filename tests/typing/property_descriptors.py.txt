from typing import assert_type

from bokeh.core.has_props import HasProps
from bokeh.core.property.descriptors import PropertyDescriptor
from bokeh.core.property.container import Dict, List
from bokeh.core.property.instance import Instance
from bokeh.core.property.nullable import Nullable
from bokeh.core.property.override import Override
from bokeh.core.property.primitive import Float, Int, String


class Child(HasProps):
    pass


class Model(HasProps):
    int_prop = Int(default=1)
    nullable_prop = Nullable(Float)
    list_prop = List(String)
    dict_prop = Dict(String, Int)
    instance_prop = Instance(Child)


model = Model()

# Instance attribute access gives property values.
assert_type(model.int_prop, int)
assert_type(model.nullable_prop, float | None)
assert_type(model.list_prop, list[str])
assert_type(model.dict_prop, dict[str, int])
assert_type(model.instance_prop, Child)

# Class attribute access gives property descriptors.
assert_type(Model.int_prop, PropertyDescriptor[int])
assert_type(Model.nullable_prop, PropertyDescriptor[float | None])

# Assignment is checked against property value types.
model.int_prop = 2
model.nullable_prop = None
model.list_prop = ["a", "b"]
model.instance_prop = Child()

model.int_prop = "2"  # mypy: assignment; pyright: reportAttributeAccessIssue
model.nullable_prop = "2"  # mypy: assignment; pyright: reportAttributeAccessIssue
model.list_prop = [1, 2]  # mypy: list-item,list-item; pyright: reportAttributeAccessIssue
model.instance_prop = Model()  # mypy: assignment; pyright: reportAttributeAccessIssue


# Overrides prefixed with an underscore don't redefine properties, so derived
# models retain types of overridden properties.

class Derived(Model):
    _int_prop = Override(default=2)
    _nullable_prop = Override(default=None)


class FurtherDerived(Derived):
    _int_prop = Override(default=lambda: 3)


derived = FurtherDerived()

assert_type(derived.int_prop, int)
assert_type(derived.nullable_prop, float | None)
assert_type(FurtherDerived.int_prop, PropertyDescriptor[int])

derived.int_prop = 4
derived.int_prop = "4"  # mypy: assignment; pyright: reportAttributeAccessIssue


# Overrides given the overridden property check types of defaults.

class TypedDerived(Model):
    _int_prop = Override(Model.int_prop, default=2)
    _nullable_prop = Override(Model.nullable_prop, default=None)
    _instance_prop = Override(Model.instance_prop, default=lambda: Child())


class TypedFurtherDerived(TypedDerived):
    _int_prop = Override(TypedDerived.int_prop, default=3)


class InvalidTypedDerived(Model):
    _int_prop = Override(Model.int_prop, default="2")  # mypy: arg-type; pyright: reportArgumentType
    _nullable_prop = Override(Model.nullable_prop, default="2")  # mypy: arg-type; pyright: reportArgumentType
    _instance_prop = Override(Model.instance_prop, default=lambda: Model())  # mypy: arg-type,return-value; pyright: reportArgumentType
    _misspelled_prop = Override(Model.misspelled_prop, default=1)  # mypy: attr-defined; pyright: reportAttributeAccessIssue


assert_type(TypedFurtherDerived().int_prop, int)
assert_type(TypedFurtherDerived.int_prop, PropertyDescriptor[int])
