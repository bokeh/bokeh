# Reproducers from issue #14814. Models are imported from their modules,
# because of issue #15763.

from collections.abc import Mapping, Sequence
from typing import Any

from bokeh.models.layouts import Column, Row
from bokeh.models.plots import Plot
from bokeh.models.ranges import Range1d
from bokeh.models.renderers.tile_renderer import TileRenderer
from bokeh.models.sources import ColumnDataSource
from bokeh.models.tools import PanTool, Toolbar

# ColumnDataSource.stream() accepts any mapping of columns.
source = ColumnDataSource()
new_data: dict[str, Sequence[Any]] = {"x": [1, 2]}
source.stream(new_data, rollover=10)
source.stream(new_data)
int_data: dict[str, list[int]] = {"x": [1, 2]}
source.stream(int_data)
float_data: Mapping[str, list[float]] = {"x": [1.0, 2.0]}
source.stream(float_data)

source.stream({"x": 1})  # mypy: dict-item; pyright: reportArgumentType

# Containers accept containers of subtypes.
rows: list[Row] = [Row()]
Column().children = rows
Column(children=rows)

# KNOWN GAP: types of these properties are declared as invariant lists and
# dicts in stubs. Properties typed in sources accept these assignments.
renderers: list[TileRenderer] = [TileRenderer()]
ranges: dict[str, Range1d] = {"y2": Range1d()}
tools: list[PanTool] = [PanTool()]
plot = Plot()
plot.renderers = renderers  # KNOWN GAP # mypy: assignment; pyright: reportAttributeAccessIssue
plot.extra_x_ranges = ranges  # KNOWN GAP # mypy: assignment; pyright: reportAttributeAccessIssue
Toolbar().tools = tools  # KNOWN GAP # mypy: assignment; pyright: reportAttributeAccessIssue
Plot(renderers=renderers, extra_x_ranges=ranges)  # KNOWN GAP # mypy: arg-type,arg-type; pyright: reportArgumentType,reportArgumentType
